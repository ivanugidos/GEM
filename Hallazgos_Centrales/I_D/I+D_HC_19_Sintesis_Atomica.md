# Función de energía potencial (Potencial de Cornell)
def V(r):
    return -K / r + sigma * r

# Función de fuerza (derivada de V)
def F(r):
    return -K / r**2 + sigma

# Bucle de evolución (gradiente descendente)
for step in range(1000):
    # Calcular fuerzas entre todos los pares
    for i in range(len(pos)):
        for j in range(i+1, len(pos)):

            r = np.linalg.norm(r_vec)
            f_mag = F(r)
            f_vec = f_mag * r_vec / r
            pos[i] += 0.01 * f_vec

print("Posiciones finales:", pos)
